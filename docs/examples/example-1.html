Пример 1: опрос датчиков на Python

Скрипт каждые 15 минут забирает последние показания и печатает тревоги.

import time
import requests

API = "https://api.zernomonitor.example/v1"
KEY = "zm_live_9f2c71"

while True:
    r = requests.get(API + "/readings", headers={"X-API-Key": KEY}, timeout=10)
    for row in r.json():
        if row["t_c"] > 25 or row["w_pct"] > 17:
            print("ALERT", row["sensor"], row["t_c"], row["w_pct"])
    time.sleep(900)
