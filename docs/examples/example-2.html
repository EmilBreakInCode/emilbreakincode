Пример 2: мини-дашборд на JavaScript

Фрагмент клиентского кода: загружает показания и рисует таблицу с подсветкой тревог.

async function load() {
  const res = await fetch("/v1/readings?limit=6", {
    headers: { "X-API-Key": KEY }
  });
  const rows = await res.json();
  table.innerHTML = rows.map(r =>
    `<tr class="${r.t_c > 25 ? 'alert' : ''}">
       <td>${r.sensor}</td><td>${r.t_c}</td><td>${r.w_pct}</td></tr>`
  ).join("");
}
setInterval(load, 900000);
